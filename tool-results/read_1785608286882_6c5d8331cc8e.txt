    41→type ConceptFormat = string;
    42→
    43→function formatTone(
    44→  format: ConceptFormat
    45→): "default" | "forest" | "gold" | "warn" | "stop" | "ok" {
    46→  const f = format.toLowerCase();
    47→  if (f.includes("carousel")) return "gold";
    48→  if (f.includes("series") || f.includes("story")) return "warn";
    49→  if (f.includes("ugc")) return "stop";
    50→  if (f.includes("event")) return "ok";
    51→  if (f.includes("reel")) return "forest";
    52→  return "default";
    53→}
    54→
    55→/** Pick an icon that signals the content format visually. */
    56→function FormatIcon({ format, className }: { format: string; className?: string }) {
    57→  const f = format.toLowerCase();
    58→  if (f.includes("carousel")) return <LayoutGrid size={14} className={className} />;
    59→  if (f.includes("series")) return <Repeat size={14} className={className} />;
    60→  if (f.includes("story")) return <Sparkles size={14} className={className} />;
    61→  if (f.includes("ugc")) return <HandHeart size={14} className={className} />;
    62→  if (f.includes("event")) return <Megaphone size={14} className={className} />;
    63→  if (f.includes("reel")) return <Film size={14} className={className} />;
    64→  return <Clapperboard size={14} className={className} />;
    65→}
