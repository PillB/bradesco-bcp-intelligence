     1→"use client";
     2→
     3→import { useState, useMemo } from "react";
     4→import { ViewHero, ViewSection, Card, Grid, DossierLinkBanner } from "../view-shell";
     5→import { Pill, Callout, StatBlock, EvidenceTag, SectionBadge } from "../ursa-brand";
     6→import { Calculator, Coffee, TrendingUp, AlertTriangle, Sparkles, RotateCcw, Info } from "lucide-react";
     7→import { Input } from "@/components/ui/input";
     8→import { Label } from "@/components/ui/label";
     9→import { Slider } from "@/components/ui/slider";
    10→import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
    11→import { cn } from "@/lib/utils";
    12→
    13→const PEN = (n: number) => {
    14→  if (!isFinite(n)) return "—";
    15→  return `S/. ${n.toFixed(2)}`;
    16→};
    17→
    18→const PRESETS = {
    19→  conservative: { subPrice: 20, costPerCup: 1.8, visitsWeek: 2.5, cupsVisit: 1, attachRate: 50, sideMargin: 4.5, cannibal: 40, avgSpendPre: 14 },
    20→  default: { subPrice: 20, costPerCup: 1.5, visitsWeek: 3, cupsVisit: 1, attachRate: 60, sideMargin: 4.5, cannibal: 30, avgSpendPre: 14 },
    21→  optimistic: { subPrice: 25, costPerCup: 1.2, visitsWeek: 3.5, cupsVisit: 1, attachRate: 70, sideMargin: 5, cannibal: 20, avgSpendPre: 16 },
    22→};
    23→
    24→type Inputs = typeof PRESETS.default;
    25→
    26→export function CalculatorView() {
    27→  const [inputs, setInputs] = useState<Inputs>(PRESETS.default);
    28→  const [weeksMonth] = useState(4.33);
    29→
    30→  const update = <K extends keyof Inputs>(key: K, value: number) => {
    31→    setInputs((prev) => ({ ...prev, [key]: value }));
    32→  };
    33→
    34→  const calc = useMemo(() => {
    35→    const { subPrice, costPerCup, visitsWeek, cupsVisit, attachRate, sideMargin, cannibal, avgSpendPre } = inputs;
    36→    const visitsMonth = visitsWeek * weeksMonth;
    37→    const cupsMonth = visitsMonth * cupsVisit;
    38→    const coffeeCost = cupsMonth * costPerCup;
    39→    const sideMarginTotal = visitsMonth * (attachRate / 100) * sideMargin;
    40→    const netProfit = subPrice + sideMarginTotal - coffeeCost;
    41→
    42→    const breakevenNoSides = costPerCup > 0 ? subPrice / costPerCup : Infinity;
    43→    const freeCapacity = costPerCup > 0 ? (subPrice + sideMarginTotal) / costPerCup : Infinity;
    44→
    45→    let cutoffAttach = NaN;
    46→    if (sideMargin > 0 && visitsMonth > 0) {
    47→      cutoffAttach = ((coffeeCost - subPrice) * 100) / (visitsMonth * sideMargin);
    48→    }
    49→
    50→    const cannibalProfit = netProfit + (cannibal / 100) * (subPrice - avgSpendPre);
    51→
    52→    let recoverMonths: number;
    53→    if (cannibalProfit < 0 && netProfit > 0) {
    54→      recoverMonths = Math.abs(cannibalProfit) / netProfit;
    55→    } else if (cannibalProfit >= 0) {
    56→      recoverMonths = 0;
    57→    } else {
    58→      recoverMonths = Infinity;
    59→    }
    60→
    61→    return { visitsMonth, cupsMonth, coffeeCost, sideMarginTotal, netProfit, breakevenNoSides, freeCapacity, cutoffAttach, cannibalProfit, recoverMonths };
    62→  }, [inputs, weeksMonth]);
    63→
    64→  // Sensitivity table data
    65→  const sensitivity = useMemo(() => {
    66→    const attachRates = [0, 20, 40, 60, 80, 100];
    67→    const visitOptions = [0.5, 1, 1.5, 2, 2.5, 3, 3.5, 4, 4.5, 5];
    68→    const rows = visitOptions.map((vw) => {
    69→      const cells = attachRates.map((ar) => {
    70→        const vm = vw * weeksMonth;
    71→        const cc = vm * inputs.cupsVisit * inputs.costPerCup;
    72→        const sm = vm * (ar / 100) * inputs.sideMargin;
    73→        const np = inputs.subPrice + sm - cc;
    74→        return { value: np, isCurrent: Math.abs(vw - inputs.visitsWeek) < 0.01 && Math.abs(ar - inputs.attachRate) < 1 };
    75→      });
    76→      return { visits: vw, cells, isCurrentRow: Math.abs(vw - inputs.visitsWeek) < 0.01 };
    77→    });
    78→    return { attachRates, rows };
    79→  }, [inputs, weeksMonth]);
    80→
    81→  // Scale projection: profit at N subscribers
    82→  const scaleProjection = useMemo(() => {
    83→    const subscriberCounts = [25, 50, 100, 200, 500];
    84→    return subscriberCounts.map((n) => ({
    85→      subscribers: n,
    86→      monthlyProfit: calc.cannibalProfit * n,
    87→      cupsPerMonth: calc.cupsMonth * n,
    88→    }));
    89→  }, [calc]);
    90→
    91→  const cellClass = (v: number) =>
    92→    v > 1 ? "bg-ursa-forest-deep/15 text-ursa-forest-deep font-semibold" : v < -1 ? "bg-ursa-terracotta/15 text-ursa-terracotta font-semibold" : "bg-muted text-muted-foreground";
    93→
    94→  return (
    95→    <>
    96→      <ViewHero
    97→        eyebrow="Module 08 · New product proposal · Interactive"
    98→        title="Ursa Mañana — Subscription Economics & Cutoff Calculator"
    99→        lede={
   100→          <>
   101→            A S/. 20/month unlimited-coffee subscription (Mon–Fri, 7–10am, in-store only, one cup per visit, no milk
   102→            drinks) is profitable on marginal-cost math because Ursa roasts its own beans. Edit every input below and
   103→            watch the profit, break-even, free-cup capacity, and sensitivity table update live. The real risk is
   104→            cannibalization, not per-cup cost — model it explicitly.
   105→          </>
   106→        }
   107→        meta={[
   108→          { label: "Pilot", value: "EXP-11 · capped 50 subscribers" },
   109→          { label: "Default", value: "S/. 20/mo · 7–10am" },
   110→          { label: "Verified cost", value: "S/. 1.20–1.80/cup (own roastery)" },
   111→        ]}
   112→      />
   113→
   114→      <ViewSection>
   115→        <DossierLinkBanner moduleId="08-subscription-economics-and-calculator" />
   116→      </ViewSection>
   117→
   118→      {/* Presets */}
   119→      <ViewSection>
   120→        <div className="flex flex-wrap items-center gap-3 mb-6">
   121→          <span className="font-label text-[0.7rem] tracking-[0.16em] uppercase text-muted-foreground">Quick scenarios:</span>
   122→          {(["conservative", "default", "optimistic"] as const).map((p) => (
   123→            <button
   124→              key={p}
   125→              onClick={() => setInputs(PRESETS[p])}
   126→              className={cn(
   127→                "px-4 py-2 rounded-full font-label text-[0.7rem] tracking-[0.12em] uppercase border transition",
   128→                "hover:-translate-y-0.5"
   129→              )}
   130→              style={{
   131→                borderColor: "var(--color-ursa-gold)",
   132→                background: "var(--color-ursa-paper)",
   133→                color: "var(--color-ursa-medium-roast)",
   134→              }}
   135→            >
   136→              {p === "conservative" && "Conservative"}
   137→              {p === "default" && "Default (recommended)"}
   138→              {p === "optimistic" && "Optimistic"}
   139→            </button>
   140→          ))}
   141→          <button
   142→            onClick={() => setInputs(PRESETS.default)}
   143→            className="ml-auto inline-flex items-center gap-1.5 px-3 py-2 rounded-full font-label text-[0.7rem] tracking-[0.12em] uppercase text-muted-foreground hover:text-ursa-dark-roast transition"
   144→          >
   145→            <RotateCcw size={13} /> Reset
   146→          </button>
   147→        </div>
   148→
   149→        <div className="grid lg:grid-cols-[1fr_1.1fr] gap-6 items-start">
   150→          {/* INPUTS */}
   151→          <Card>
   152→            <h3 className="font-display text-xl font-semibold text-ursa-dark-roast mt-0 mb-1 flex items-center gap-2">
   153→              <Calculator size={20} className="text-ursa-gold" /> Edit the inputs
   154→            </h3>
   155→            <p className="text-[0.85rem] text-muted-foreground mb-5">Drag sliders or type values. Everything recalculates instantly.</p>
   156→
   157→            <div className="space-y-5">
   158→              {/* Subscription price */}
   159→              <NumberField label="Subscription price (monthly)" value={inputs.subPrice} onChange={(v) => update("subPrice", v)} min={5} max={60} step={1} prefix="S/. " hint="The price a subscriber pays per month." />
   160→              {/* Cost per cup */}
   161→              <NumberField label="Marginal cost per cup" value={inputs.costPerCup} onChange={(v) => update("costPerCup", v)} min={0.3} max={4} step={0.05} prefix="S/. " hint="Bean + water + energy + cup. Own-roastery benchmark: S/. 1.20–1.80." />
   162→
   163→              <SliderField label="Visits per week" value={inputs.visitsWeek} onChange={(v) => update("visitsWeek", v)} min={0.5} max={5} step={0.5} display={`${inputs.visitsWeek.toFixed(1)} visits`} hint="How often a subscriber comes in during the 7–10am window." />
   164→              <SliderField label="Cups per visit" value={inputs.cupsVisit} onChange={(v) => update("cupsVisit", v)} min={1} max={3} step={1} display={`${inputs.cupsVisit}`} hint="Capped at 1 in the pilot to prevent abuse." />
   165→
   166→              <SliderField label="Side attach rate" value={inputs.attachRate} onChange={(v) => update("attachRate", v)} min={0} max={100} step={5} display={`${inputs.attachRate}%`} hint="% of visits where the subscriber also buys a cookie, pastry, or empanada. This is the lever that makes the model work." />
   167→              <NumberField label="Side margin (per attach)" value={inputs.sideMargin} onChange={(v) => update("sideMargin", v)} min={1} max={12} step={0.25} prefix="S/. " hint="Profit margin on the accompanying food item, not revenue." />
   168→
   169→              <SliderField label="Cannibalization rate" value={inputs.cannibal} onChange={(v) => update("cannibal", v)} min={0} max={100} step={5} display={`${inputs.cannibal}%`} hint="% of subscribers who would have paid full price anyway. The biggest risk." />
   170→              <NumberField label="Avg. pre-subscription spend" value={inputs.avgSpendPre} onChange={(v) => update("avgSpendPre", v)} min={6} max={30} step={0.5} prefix="S/. " hint="What a cannibalized customer used to spend per visit, before subscribing." />
   171→            </div>
   172→
   173→            <Callout tone="warn" title="The side-pushing strategy is the whole game">
   174→              <p className="m-0">
   175→                Unlimited coffee only works if subscribers buy accompanying food at a healthy rate. The café should
   176→                actively push sides — cookies, the financier de pera, the empanada, seasonal pastries — through pairing
   177→                cards, bear-recommends prompts, and a visible morning pastry case. This calculator assumes you do that;
   178→                if the attach rate stays below 40%, kill the pilot (see stop rule).
   179→              </p>
   180→            </Callout>
   181→          </Card>
   182→
   183→          {/* OUTPUTS */}
   184→          <div className="space-y-5">
   185→            {/* Headline profit */}
   186→            <Card highlight className="bg-gradient-to-br from-ursa-paper to-ursa-cream">
   187→              <div className="flex items-center justify-between mb-1">
   188→                <span className="font-label text-[0.7rem] tracking-[0.18em] uppercase text-ursa-gold">Net profit per subscriber / month</span>
   189→                <EvidenceTag status="verified" />
   190→              </div>
   191→              <div className={cn("font-display text-5xl font-semibold leading-none mb-2", calc.netProfit >= 0 ? "text-ursa-forest-deep" : "text-ursa-terracotta")}>
   192→                {PEN(calc.netProfit)}
   193→              </div>
   194→              <p className="text-[0.85rem] text-muted-foreground m-0">
   195→                {calc.netProfit >= 0
   196→                  ? "Profitable at these assumptions. Scroll down for the sensitivity table and cannibalization-adjusted view."
   197→                  : "Not profitable at these assumptions. Raise the price, raise the attach rate, or lower the cost per cup."}
   198→              </p>
   199→            </Card>
   200→
   201→            {/* Key metrics grid */}
   202→            <Grid cols={2}>
   203→              <MiniMetric icon={<Coffee size={16} />} label="Visits / month" value={calc.visitsMonth.toFixed(1)} tone="neutral" />
   204→              <MiniMetric icon={<Coffee size={16} />} label="Cups / month" value={calc.cupsMonth.toFixed(0)} tone="neutral" />
   205→              <MiniMetric icon={<TrendingUp size={16} />} label="Coffee cost / month" value={PEN(calc.coffeeCost)} tone="terracotta" />
   206→              <MiniMetric icon={<Sparkles size={16} />} label="Side margin / month" value={PEN(calc.sideMarginTotal)} tone="forest" />
   207→              <MiniMetric icon={<Info size={16} />} label="Break-even (no sides)" value={isFinite(calc.breakevenNoSides) ? `${calc.breakevenNoSides.toFixed(1)} cups` : "∞"} tone="neutral" />
   208→              <MiniMetric icon={<Info size={16} />} label="Free-cup capacity" value={isFinite(calc.freeCapacity) ? `${calc.freeCapacity.toFixed(1)} cups` : "∞"} tone="forest" />
   209→            </Grid>
   210→
   211→            {/* Cutoff attach */}
   212→            <Card>
   213→              <div className="flex items-center justify-between mb-2">
   214→                <h4 className="font-display text-base font-semibold text-ursa-dark-roast m-0">Minimum attach rate to break even</h4>
   215→                <Pill tone={isFinite(calc.cutoffAttach) && calc.cutoffAttach <= inputs.attachRate ? "ok" : "warn"}>
   216→                  {isFinite(calc.cutoffAttach) ? `${calc.cutoffAttach.toFixed(0)}%` : "n/a"}
   217→                </Pill>
   218→              </div>
   219→              <p className="text-[0.85rem] text-muted-foreground m-0">
   220→                {isFinite(calc.cutoffAttach)
   221→                  ? calc.cutoffAttach <= inputs.attachRate
   222→                    ? `You are above the break-even attach rate by ${(inputs.attachRate - calc.cutoffAttach).toFixed(0)} points. Headroom is healthy.`
   223→                    : `You are below break-even by ${(calc.cutoffAttach - inputs.attachRate).toFixed(0)} points. Push sides harder or raise the price.`
   224→                  : "Cannot compute — set a side margin and visit frequency."}
   225→              </p>
   226→            </Card>
   227→
   228→            {/* Cannibalization */}
   229→            <Card className="bg-ursa-foam">
   230→              <h4 className="font-display text-base font-semibold text-ursa-dark-roast mt-0 mb-2 flex items-center gap-2">
   231→                <AlertTriangle size={16} className="text-ursa-gold" /> Cannibalization-adjusted profit
   232→              </h4>
   233→              <div className="flex items-baseline gap-3 mb-2">
   234→                <span className={cn("font-display text-3xl font-semibold", calc.cannibalProfit >= 0 ? "text-ursa-forest-deep" : "text-ursa-terracotta")}>
   235→                  {PEN(calc.cannibalProfit)}
   236→                </span>
   237→                <span className="text-[0.8rem] text-muted-foreground">per subscriber / month, blended</span>
   238→              </div>
   239→              <p className="text-[0.85rem] text-muted-foreground m-0">
   240→                {calc.cannibalProfit >= 0
   241→                  ? `Even accounting for ${inputs.cannibal}% of subscribers who would have paid full price, the blended profit is positive.`
   242→                  : `At ${inputs.cannibal}% cannibalization, the blended profit is negative. Reduce cannibalization (target new customers, not existing regulars) or raise the price.`}
   243→              </p>
   244→              <div className="mt-3 pt-3 border-t border-ursa-line-soft flex items-center justify-between">
   245→                <span className="font-label text-[0.7rem] tracking-[0.12em] uppercase text-muted-foreground">Recovery time</span>
   246→                <span className="font-display text-base font-semibold text-ursa-dark-roast">
   247→                  {isFinite(calc.recoverMonths) ? (calc.recoverMonths === 0 ? "Already positive" : `${calc.recoverMonths.toFixed(1)} months`) : "Never (kill the pilot)"}
   248→                </span>
   249→              </div>
   250→            </Card>
   251→          </div>
   252→        </div>
   253→      </ViewSection>
   254→
   255→      {/* Tabs: sensitivity + scale + assumptions */}
   256→      <ViewSection badge="Deeper analysis" title="Sensitivity, scale, and the honest assumptions">
   257→        <Tabs defaultValue="sensitivity">
   258→          <TabsList className="bg-muted overflow-x-auto ursa-scroll justify-start sm:justify-center flex-nowrap max-w-full">
   259→            <TabsTrigger value="sensitivity">Sensitivity table</TabsTrigger>
   260→            <TabsTrigger value="scale">Scale projection</TabsTrigger>
   261→            <TabsTrigger value="assumptions">Honest assumptions</TabsTrigger>
   262→            <TabsTrigger value="sides">Side-pushing playbook</TabsTrigger>
   263→          </TabsList>
   264→
   265→          <TabsContent value="sensitivity" className="mt-6">
   266→            <Card>
   267→              <p className="text-[0.9rem] text-muted-foreground mb-4">
   268→                Net profit per subscriber/month across visit frequency (rows) and side attach rate (columns). The
   269→                highlighted cell is your current scenario. Green is profitable; red is not.
   270→              </p>
   271→              <div className="overflow-x-auto ursa-scroll">
   272→                <table className="w-full border-collapse text-[0.85rem]">
   273→                  <thead>
   274→                    <tr>
   275→                      <th className="text-left p-2 font-label text-[0.7rem] tracking-[0.1em] uppercase text-muted-foreground border-b border-ursa-line">
   276→                        Visits/wk ↓ / Attach →
   277→                      </th>
   278→                      {sensitivity.attachRates.map((ar) => (
   279→                        <th key={ar} className="p-2 font-label text-[0.7rem] tracking-[0.1em] uppercase text-muted-foreground border-b border-ursa-line text-center">
   280→                          {ar}%
   281→                        </th>
   282→                      ))}
   283→                    </tr>
   284→                  </thead>
   285→                  <tbody>
   286→                    {sensitivity.rows.map((row) => (
   287→                      <tr key={row.visits} className={row.isCurrentRow ? "bg-ursa-gold/10" : ""}>
   288→                        <td className="p-2 font-label text-[0.8rem] text-ursa-dark-roast border-b border-ursa-line-soft">
   289→                          {row.visits.toFixed(1)}
   290→                        </td>
   291→                        {row.cells.map((c, i) => (
   292→                          <td key={i} className={cn("p-2 text-center border-b border-ursa-line-soft", cellClass(c.value), c.isCurrent && "outline outline-2 outline-ursa-gold outline-offset-[-2px]")}>
   293→                            {c.value.toFixed(1)}
   294→                          </td>

   296→                      </tr>

   298→                  </tbody>
   299→                </table>

   308→                to validate before scaling.
   309→              </p>
   310→              <div className="space-y-3">
   311→                {scaleProjection.map((s) => {
   312→                  const isPilot = s.subscribers === 50;
   313→                  const pctOfMax = (s.monthlyProfit / (scaleProjection[scaleProjection.length - 1].monthlyProfit || 1)) * 100;
   314→                  return (
